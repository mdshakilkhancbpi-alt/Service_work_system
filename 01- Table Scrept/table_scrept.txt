-- 1. ROLES & USERS
--------------------------------------------------------
CREATE TABLE XH_USERS (
    ID               NUMBER(10,0),
    USER_NAME        VARCHAR2(50) NOT NULL ENABLE,  
    USER_PASSWORD    VARCHAR2(50),
    FULL_NAME        VARCHAR2(100) NOT NULL,
    ROLE             VARCHAR2(20) NOT NULL CHECK (ROLE IN ('ENGINEER', 'DISPATCHER', 'MANAGER')),
    PHONE            VARCHAR2(20), 
    STATUS           VARCHAR2(1) DEFAULT 'Y',
    LAST_LOGIN_DATE  DATE,  
    CREATED_BY       VARCHAR2(40), 
    MODIFIED_BY      VARCHAR2(40), 
    CREATED_DATE     TIMESTAMP (6) WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL,
    MODIFIED_DATE    TIMESTAMP (6) WITH TIME ZONE
);

ALTER TABLE XH_USERS ADD CONSTRAINT PK_XH_USER_ID PRIMARY KEY (ID);
ALTER TABLE XH_USERS ADD CONSTRAINT UK_XH_USER_NAME UNIQUE (USER_NAME);

CREATE OR REPLACE TRIGGER "XH_USERS_TRG"
   BEFORE INSERT OR UPDATE ON XH_USERS
   FOR EACH ROW
BEGIN
   IF INSERTING THEN
      IF :NEW.ID IS NULL THEN
         SELECT NVL(MAX(ID), 0) + 1 INTO :NEW.ID FROM XH_USERS;
      END IF;
      :NEW.CREATED_DATE := CURRENT_TIMESTAMP;
      :NEW.CREATED_BY   := NVL(V('APP_USER'), USER);
   END IF;
   IF UPDATING THEN
      :NEW.MODIFIED_DATE := CURRENT_TIMESTAMP;
      :NEW.MODIFIED_BY   := NVL(V('APP_USER'), USER);
   END IF;
END;

--------------------------------------------------------
-- WORK ORDERS TRIGGER (CREATED AFTER AUDIT TABLE)
--------------------------------------------------------
CREATE OR REPLACE TRIGGER "XH_WORK_ORDERS_TRG"
   BEFORE INSERT OR UPDATE ON XH_WORK_ORDERS
   FOR EACH ROW
BEGIN
   IF INSERTING THEN
      IF :NEW.ID IS NULL THEN
         SELECT NVL(MAX(ID), 0) + 1 INTO :NEW.ID FROM XH_WORK_ORDERS;
      END IF;
      :NEW.CREATED_DATE := CURRENT_TIMESTAMP;
      :NEW.CREATED_BY   := NVL(V('APP_USER'), USER);
   END IF;
   
   IF UPDATING THEN
      :NEW.MODIFIED_DATE := CURRENT_TIMESTAMP;
      :NEW.MODIFIED_BY   := NVL(V('APP_USER'), USER);

      -- RULE 1: STRICT LIFECYCLE STATE TRANSITION ENFORCEMENT
      IF :OLD.STATUS != :NEW.STATUS THEN
         IF :OLD.STATUS = 'NEW' AND :NEW.STATUS NOT IN ('ASSIGNED', 'CANCELLED') THEN
            RAISE_APPLICATION_ERROR(-20001, 'Invalid Transition: NEW status can only move to ASSIGNED or CANCELLED.');
         ELSIF :OLD.STATUS = 'ASSIGNED' AND :NEW.STATUS NOT IN ('IN PROGRESS', 'CANCELLED') THEN
            RAISE_APPLICATION_ERROR(-20002, 'Invalid Transition: ASSIGNED status can only move to IN PROGRESS or CANCELLED.');
         ELSIF :OLD.STATUS = 'IN PROGRESS' AND :NEW.STATUS NOT IN ('ON HOLD', 'COMPLETED', 'CANCELLED') THEN
            RAISE_APPLICATION_ERROR(-20003, 'Invalid Transition: IN PROGRESS status can move to ON HOLD, COMPLETED, or CANCELLED.');
         ELSIF :OLD.STATUS = 'ON HOLD' AND :NEW.STATUS NOT IN ('IN PROGRESS', 'CANCELLED') THEN
            RAISE_APPLICATION_ERROR(-20004, 'Invalid Transition: ON HOLD status can only return to IN PROGRESS or CANCELLED.');
         ELSIF :OLD.STATUS IN ('COMPLETED', 'CLOSED', 'CANCELLED') THEN
            RAISE_APPLICATION_ERROR(-20005, 'Invalid Transition: Terminal status cannot be modified directly.');
         END IF;

         -- RULE 6: AUTOMATIC AUDIT LOGGING FOR STATUS CHANGE
         INSERT INTO XH_WORK_ORDER_AUDITS (
            XH_WORK_ORDER_ID, OLD_STATUS, NEW_STATUS, ACTION_SOURCE, CREATED_BY, CREATED_DATE
         ) VALUES (
            :NEW.ID, :OLD.STATUS, :NEW.STATUS,
            CASE WHEN V('APP_USER') IS NOT NULL THEN 'APEX_APP' ELSE 'SQL_DIRECT' END,
            NVL(V('APP_USER'), USER), CURRENT_TIMESTAMP
         );
      END IF;
   END IF;
END;
/
