--------------------------------------------------------
-- SYSTEM / SYS USER SETUP SCRIPT FOR FIELDSERVICE
--------------------------------------------------------

-- 1. Create User
CREATE USER FIELDSERVICE 
IDENTIFIED BY "ADMIN321#";

-- 2. Grant Roles & Privileges
GRANT DBA TO FIELDSERVICE;
GRANT CONNECT, RESOURCE TO FIELDSERVICE;

-- 3. Set Tablespace & Quota
ALTER USER FIELDSERVICE DEFAULT TABLESPACE USERS;
ALTER USER FIELDSERVICE QUOTA UNLIMITED ON USERS;

-- 4. Save Changes
COMMIT;
